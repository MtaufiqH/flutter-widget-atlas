ThemeExtension (custom tokens)

ThemeData has no slot for "success" or "warning" colours. A ThemeExtension adds your own tokens to the theme. They are available through Theme.of(context), they change with light and dark mode, and they animate when the theme changes.

Use it for

App-specific tokens: success and warning colours, a brand gradient, custom shadows, spacing that is part of a design system.

Watch out for

A ThemeExtension must implement both copyWith and lerp. Register one const instance per theme (light and dark), and read it with a null check. A small extension on BuildContext keeps the call sites short.

Example code

@immutable
class AppTokens extends ThemeExtension<AppTokens> {
  const AppTokens({
    required this.success,
    required this.warning,
  });
  final Color success;
  final Color warning;

  @override
  AppTokens copyWith({Color? success, Color? warning}) =>
      AppTokens(
        success: success ?? this.success,
        warning: warning ?? this.warning,
      );

  @override
  AppTokens lerp(ThemeExtension<AppTokens>? other, double t) {
    if (other is! AppTokens) return this;
    return AppTokens(
      success: Color.lerp(success, other.success, t)!,
      warning: Color.lerp(warning, other.warning, t)!,
    );
  }
}

// register one per theme
ThemeData(
  extensions: const [
    AppTokens(
      success: Color(0xFF1B7F4B),
      warning: Color(0xFFB26A00),
    ),
  ],
)

// helper, once
extension ThemeX on BuildContext {
  ColorScheme get colors => Theme.of(this).colorScheme;
  TextTheme get text => Theme.of(this).textTheme;
  AppTokens get tokens => Theme.of(this).extension<AppTokens>()!;
}

// use
Text('Paid', style: TextStyle(color: context.tokens.success))
